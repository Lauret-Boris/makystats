VECM / Johansen

Vector error-correction models with Johansen cointegration testing. Rank selection can use MHM p-values or critical values.

from makystats import vecm, select_coint_rank, summarize_johansen

# Rank via MHM p-values
rank = select_coint_rank(data, k_ar_diff=1, test_case="3")

# Estimate VECM (det_case uses the package coding, not statsmodels names)
res = vecm(data, k_ar_diff=1, coint_rank=1, det_case="3a").fit()
print(res.summary())

# Critical-value summary
print(summarize_johansen(data, k_ar_diff=1, test_case="3"))
