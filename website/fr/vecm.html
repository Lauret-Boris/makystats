VECM / Johansen

Modèles à correction d’erreur vectorielle avec test de cointégration de Johansen. La sélection du rang peut utiliser les p-values MHM ou les valeurs critiques.

from makystats import vecm, select_coint_rank, summarize_johansen

rank = select_coint_rank(data, k_ar_diff=1, test_case="3")
res = vecm(data, k_ar_diff=1, coint_rank=1, det_case="3a").fit()
print(res.summary())
print(summarize_johansen(data, k_ar_diff=1, test_case="3"))
